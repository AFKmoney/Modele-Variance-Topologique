    25→# =====================================================================
    26→# FLOPs Counter (hooks into numpy operations)
    27→# =====================================================================
    28→
    29→class FLOPsCounter:
    30→    """Approximate FLOPs counter based on operation shapes."""
    31→    def __init__(self):
    32→        self.flops = 0
    33→        self.ops = 0
    34→
    35→    def add_matmul(self, M, N, K):
    36→        """Matrix multiply C[M,N] = A[M,K] @ B[K,N] → 2*M*N*K FLOPs"""
    37→        self.flops += 2 * M * N * K
    38→        self.ops += 1
    39→
    40→    def add_einsum(self, equation, *shapes):
    41→        """Estimate FLOPs for einsum from shape sizes."""
    42→        # Parse output size
    43→        out_size = 1
    44→        for s in shapes:
    45→            for d in s:
    46→                out_size *= d
    47→        # Rough estimate: output_size * average_input_size
    48→        in_size = sum(math.prod(s) for s in shapes)
    49→        self.flops += max(out_size, in_size * 2)
    50→        self.ops += 1
    51→
    52→    def reset(self):
    53→        self.flops = 0
    54→        self.ops = 0

    73→def bench_metric_tensor(N: int, n_calls: int = 50) -> Dict:
    74→    """Benchmark: MetricTensor operations (Christoffel, geodesic accel)."""
    75→    from mvt.config import MVTConfig
    76→    from mvt.core.metric_tensor import MetricTensor
    77→
    78→    cfg = MVTConfig(ambient_dim=N, intrinsic_dim=max(N//2, 16))
    79→    metric = MetricTensor(cfg)
    80→    q = np.random.randn(N)
    81→    dq = np.random.randn(N)
    82→
    83→    # Warmup
    84→    for _ in range(3):
    85→        metric.christoffel_symbols(q)
    86→        metric.geodesic_acceleration(q, dq)
    87→        metric.scalar_curvature(q)
    88→
    89→    # Benchmark Christoffel symbols
    90→    t0 = time.perf_counter()
    91→    for _ in range(n_calls):
    92→        metric.christoffel_symbols(q)
    93→    t_christoffel = (time.perf_counter() - t0) / n_calls
    94→
    95→    # Benchmark geodesic acceleration
    96→    t0 = time.perf_counter()
    97→    for _ in range(n_calls):
    98→        metric.geodesic_acceleration(q, dq)
    99→    t_geodesic = (time.perf_counter() - t0) / n_calls
   100→
   101→    # Benchmark scalar curvature
   102→    t0 = time.perf_counter()
   103→    for _ in range(n_calls):
   104→        metric.scalar_curvature(q)
   105→    t_curvature = (time.perf_counter() - t0) / n_calls
   106→
   107→    # FLOPs estimation for Christoffel symbols
   108→    # dG computation: ~3 * N^3 (sin, cos, multiply)
   109→    # einsum 'kl,lij->kij': 2 * N^3 * N = 2 * N^4
   110→    flops_christoffel = 3 * N**3 + 2 * N**4
   111→    # geodesic accel: einsum 'kij,i,j->k': 2 * N^3
   112→    flops_geodesic = 2 * N**3
   113→
   114→    return {
   115→        "N": N,
   116→        "christoffel_ms": t_christoffel * 1000,
   117→        "geodesic_ms": t_geodesic * 1000,
   118→        "curvature_ms": t_curvature * 1000,
   119→        "flops_christoffel": flops_christoffel,
   120→        "flops_geodesic": flops_geodesic,
   121→        "gflops_christoffel": flops_christoffel / t_christoffel / 1e9 if t_christoffel > 0 else 0,
   122→        "gflops_geodesic": flops_geodesic / t_geodesic / 1e9 if t_geodesic > 0 else 0,
   123→    }

   126→def bench_rk4_integration(N: int, n_steps: int = 200, n_calls: int = 10) -> Dict:
   127→    """Benchmark: Full RK4 integration step."""
   128→    from mvt.config import MVTConfig
   129→    from mvt.core.metric_tensor import MetricTensor
   130→    from mvt.lagrangian.semantic_lagrangian import SemanticLagrangian
   131→    from mvt.lagrangian.integrator import LagrangianIntegrator
   132→
   133→    cfg = MVTConfig(ambient_dim=N, intrinsic_dim=max(N//2, 16), num_rk4_steps=n_steps)
   134→    metric = MetricTensor(cfg)
   135→    lagrangian = SemanticLagrangian(cfg, metric)
   136→    integrator = LagrangianIntegrator(cfg, lagrangian)
   137→
   138→    q0 = np.random.randn(N) * 0.01
   139→    dq0 = np.random.randn(N) * 0.01
   140→
   141→    # Warmup
   142→    integrator.integrate(q0, dq0, num_steps=20)
   143→
   144→    # Benchmark
   145→    t0 = time.perf_counter()
   146→    for _ in range(n_calls):
   147→        integrator.integrate(q0, dq0, num_steps=n_steps)
   148→    t_total = (time.perf_counter() - t0) / n_calls
   149→
   150→    t_per_step = t_total / n_steps
   151→
   152→    # FLOPs per RK4 step:
   153→    # 4 evaluations of derivatives
   154→    # Each evaluation: euler_lagrange_rhs = geodesic_accel + potential + damping
   155→    # geodesic_accel = christoffel + einsum = ~2*N^4 + 2*N^3
   156→    # potential: ~N
   157→    # So per step: 4 * (2*N^4 + 2*N^3 + N) ≈ 8*N^4
   158→    flops_per_step = 8 * N**4 + 8 * N**3
   159→    flops_total = flops_per_step * n_steps
   160→
   161→    return {
   162→        "N": N,
   163→        "n_steps": n_steps,
   164→        "total_time_ms": t_total * 1000,
   165→        "time_per_step_ms": t_per_step * 1000,
   166→        "flops_per_step": flops_per_step,
   167→        "flops_total": flops_total,
   168→        "gflops_sec": flops_total / t_total / 1e9 if t_total > 0 else 0,
   169→        "steps_per_sec": n_steps / t_total if t_total > 0 else 0,
   170→    }

   173→def bench_moe_forward(d_model: int, n_experts: int, top_k: int,
   174→                      n_layers: int, batch: int = 4, seq_len: int = 32) -> Dict:
   175→    """Benchmark: MoE-MVT forward pass on CPU."""
   176→    import torch
   177→    from mvt.edt.moe_model import MoEMVT, MoEMVTConfig
   178→
   179→    cfg = MoEMVTConfig(
   180→        d_model=d_model,
   181→        n_layers=n_layers,
   182→        n_experts=n_experts,
   183→        top_k=top_k,
   184→        d_ff=d_model * 2,
   185→        max_seq_len=seq_len,
   186→    )
   187→
   188→    model = MoEMVT(cfg)
   189→    model.eval()
   190→
   191→    tokens = torch.randint(0, cfg.vocab_size, (batch, seq_len))
   192→
   193→    # Warmup
   194→    with torch.no_grad():
   195→        for _ in range(3):
   196→            model(tokens)
   197→
   198→    # Benchmark
   199→    t0 = time.perf_counter()
   200→    n_calls = 20
   201→    with torch.no_grad():
   202→        for _ in range(n_calls):
   203→            model(tokens)
   204→    t_forward = (time.perf_counter() - t0) / n_calls
   205→
   206→    # Parameter count
   207→    total_p, active_p = model.count_params()
   208→
   209→    # FLOPs estimation per token per layer:
   210→    # Attention: 4 * d_model^2 * seq_len (QKV) + 2 * d_model * seq_len^2 (attn weights)
   211→    # MoE router: d_model * n_experts
   212→    # MoE experts (top_k): top_k * 2 * d_model * d_ff
   213→    attn_flops = 4 * d_model**2 * seq_len + 2 * d_model * seq_len**2
   214→    moe_flops = d_model * n_experts + top_k * 2 * d_model * cfg.d_ff
   215→    flops_per_token_layer = attn_flops + moe_flops
   216→    flops_per_sample = flops_per_token_layer * seq_len * n_layers
   217→    flops_total = flops_per_sample * batch
   218→
   219→    return {
   220→        "d_model": d_model,
   221→        "n_experts": n_experts,
   222→        "top_k": top_k,
   223→        "n_layers": n_layers,
   224→        "batch": batch,
   225→        "seq_len": seq_len,
   226→        "total_params": total_p,
   227→        "active_params_per_token": active_p,
   228→        "sparsity": 1 - active_p / total_p,
   229→        "forward_ms": t_forward * 1000,
   230→        "flops_per_sample": flops_per_sample,
   231→        "gflops_sec": flops_total / t_forward / 1e9 if t_forward > 0 else 0,
   232→        "samples_per_sec": 1.0 / t_forward if t_forward > 0 else 0,
   233→        "tokens_per_sec": batch * seq_len / t_forward if t_forward > 0 else 0,
   234→    }

   237→def bench_edt_phase1(d_model: int, n_experts: int, n_layers: int,
   238→                     steps_per_expert: int = 50, batch: int = 8) -> Dict:
   239→    """Benchmark: EDT Phase 1 (expert independent training) cost per expert."""
   240→    import torch
   241→    import torch.nn.functional as F
   242→    from mvt.edt.moe_model import MoEMVT, MoEMVTConfig, TopoExpert
   243→
   244→    d_ff = d_model * 2
   245→
   246→    expert = TopoExpert(d_model, d_ff)
   247→    opt = torch.optim.AdamW(expert.parameters(), lr=1e-3)
   248→
   249→    h_in = torch.randn(batch, 32, d_model)
   250→    h_target = torch.randn(batch, 32, d_model)
   251→
   252→    # Warmup
   253→    for _ in range(3):
   254→        opt.zero_grad()
   255→        out = expert(h_in)
   256→        loss = F.mse_loss(out, h_target)
   257→        loss.backward()
   258→        opt.step()
   259→
   260→    # Benchmark
   261→    t0 = time.perf_counter()
   262→    for _ in range(steps_per_expert):
   263→        opt.zero_grad()
   264→        out = expert(h_in)
   265→        loss = F.mse_loss(out, h_target)
   266→        loss.backward()
   267→        opt.step()
   268→    t_expert = time.perf_counter() - t0
   269→
   270→    t_per_step = t_expert / steps_per_expert
   271→    total_experts = n_experts * n_layers
   272→
   273→    # FLOPs per step: forward 2*d_model*d_ff + backward ~3x forward
   274→    flops_fwd = 2 * d_model * d_ff * batch * 32
   275→    flops_bwd = 3 * flops_fwd
   276→    flops_per_step = flops_fwd + flops_bwd
   277→
   278→    return {
   279→        "d_model": d_model,
   280→        "d_ff": d_ff,
   281→        "n_experts_per_layer": n_experts,
   282→        "n_layers": n_layers,
   283→        "total_experts": total_experts,
   284→        "steps_per_expert": steps_per_expert,
   285→        "batch": batch,
   286→        "time_per_step_ms": t_per_step * 1000,
   287→        "time_per_expert_s": t_expert,
   288→        "time_all_experts_s": t_expert * total_experts,
   289→        "flops_per_step": flops_per_step,
   290→        "gflops_sec": flops_per_step / t_per_step / 1e9 if t_per_step > 0 else 0,
   291→    }

   324→# MVT Scaling Calculator

   327→def calculate_mvt_scaling(target_params: float, bench_data: Dict) -> Dict:
   328→    """
   329→    Calculate Chinchilla-adapted scaling for MVT.
   330→
   331→    Key insight: MVT uses integration steps instead of tokens.
   332→    Each step costs O(N^4) FLOPs (vs O(N) per token in transformers).
   333→    But geometry is more expressive per parameter → lower data ratio needed.
   334→    """
   335→    # Architecture to parameter mapping
   336→    # For MoE-MVT: params ≈ n_layers * (attn_params + n_experts * expert_params) + embedding
   337→    # expert_params = 2 * d_model * d_ff
   338→    # attn_params = 4 * d_model^2 (QKV) + d_model^2 (out_proj) + 3*d_model (norm)
   339→    # embedding = vocab_size * d_model
   340→
   341→    # Estimate N (d_model) from target params
   342→    # Rough: total_params ≈ n_layers * n_experts * 2 * d_model^2 (dominant term)
   343→    # → d_model ≈ sqrt(target_params / (n_layers * n_experts * 2))
   344→
   345→    # Try different architectures to hit target
   346→    configs = []
   347→
   348→    for n_layers in [6, 8, 12, 16, 24]:
   349→        for n_experts in [8, 16, 32, 64]:
   350→            for d_ff_ratio in [2, 4]:
   351→                d_ff_base = d_ff_ratio
   352→                # Solve: target ≈ n_layers * n_experts * 2 * d_model * d_ff_base * d_model
   353→                #        + n_layers * 5 * d_model^2 (attention)
   354→                #        + vocab * d_model (embedding)
   355→                # Simplify: d_model^2 * (n_layers * n_experts * 2 * d_ff_base + n_layers * 5 + vocab)
   356→
   357→                vocab = 32000
   358→                coeff = n_layers * n_experts * 2 * d_ff_base + n_layers * 5 + vocab
   359→                d_model = math.sqrt(target_params / coeff)
   360→                d_model = int(round(d_model))
   361→
   362→                if d_model < 32 or d_model > 8192:
   363→                    continue
   364→
   365→                d_ff = d_ff_base * d_model
   366→                top_k = max(2, n_experts // 8)
   367→
   368→                # Count actual params
   369→                # Embedding + position
   370→                embed_params = vocab * d_model + 128 * d_model
   371→                # Attention per layer
   372→                attn_params = n_layers * (4 * d_model * d_model + d_model * d_model + 2 * d_model)
   373→                # Experts per layer
   374→                expert_params = n_layers * n_experts * (d_model * d_ff + d_ff * d_model)
   375→                # Router per layer
   376→                router_params = n_layers * (d_model * n_experts)
   377→                # Norms
   378→                norm_params = n_layers * 3 * 2 * d_model  # 3 norms, 2 (gamma + beta)
   379→                total = embed_params + attn_params + expert_params + router_params + norm_params
   380→
   381→                error = abs(total - target_params) / target_params
   382→                if error > 0.3:  # Skip configs that are way off
   383→                    continue
   384→
   385→                configs.append({
   386→                    "n_layers": n_layers,
   387→                    "n_experts": n_experts,
   388→                    "d_model": d_model,
   389→                    "d_ff": d_ff,
   390→                    "top_k": top_k,
   391→                    "total_params": total,
   392→                    "active_params_ratio": top_k / n_experts,
   393→                    "error_pct": error * 100,
   394→                })
   395→
   396→    # Pick best config (closest to target)
   397→    configs.sort(key=lambda x: abs(x["total_params"] - target_params))
   398→    best = configs[0] if configs else None
   399→
   400→    if best is None:
   401→        return {"error": "Could not find config matching target params"}
   402→
   403→    # Calculate scaling metrics
   404→    d_model = best["d_model"]
   405→    n_steps = 200  # default integration steps from MVTConfig
   406→
   407→    # FLOPs per integration step (RK4 with Christoffel)
   408→    # 4 deriv evaluations × (2*N^4 + 2*N^3) per eval
   409→    flops_per_step = 8 * d_model**4 + 8 * d_model**3
   410→
   411→    # FLOPs per sample (one full integration trajectory)
   412→    flops_per_sample = flops_per_step * n_steps
   413→
   414→    # For MoE forward (if using MoE-MVT):
   415→    moe_flops_per_token = (4 * d_model**2 + 2 * d_model * 128  # attention
   416→                           + best["top_k"] * 2 * d_model * best["d_ff"])  # MoE
   417→    moe_flops_per_sample = moe_flops_per_token * 128 * best["n_layers"]
   418→
   419→    # Combined: geometry + MoE
   420→    combined_flops_per_sample = flops_per_sample + moe_flops_per_sample
   421→
   422→    # PGSG reduction (60% of backward)
   423→    backward_reduction = 0.6
   424→    effective_flops = combined_flops_per_sample * (1 + (1 - backward_reduction))
   425→
   426→    # MVT Chinchilla ratio
   427→    # Transformer: D/N ≈ 20 tokens/param
   428→    # MVT: Each sample is more expensive but more expressive
   429→    # Efficiency multiplier: geometry encodes more structure per param
   430→    expr_mult = 3.5  # continuous geometry is ~3.5x more expressive per param
   431→    cost_mult = combined_flops_per_sample / (6 * target_params)  # relative to transformer FLOPs per token
   432→
   433→    # Adjusted ratio
   434→    mvt_ratio = 20 * expr_mult / (cost_mult / (6 * target_params))
   435→    # Simpler formulation: 
   436→    # optimal_samples = (C_transformer / C_mvt_per_sample) * expr_mult
   437→    # where C_transformer = 6 * N * 20*N = 120 * N^2
   438→    C_transformer = 120 * target_params  # total FLOPs for Chinchilla transformer
   439→    optimal_samples = C_transformer / effective_flops * expr_mult
   440→    actual_ratio = optimal_samples / target_params
   441→
   442→    # Time estimates based on benchmark data
   443→    if "rk4" in bench_data:
   444→        # Extrapolate from benchmark
   445→        bench_N = bench_data["rk4"]["N"]
   446→        bench_time_per_step = bench_data["rk4"]["time_per_step_ms"] / 1000
   447→        scale_factor = (d_model / bench_N) ** 4  # FLOPs scale as N^4
   448→        est_time_per_step = bench_time_per_step * scale_factor
   449→        est_time_per_sample = est_time_per_step * n_steps
   450→
   451→        time_per_sample_s = est_time_per_step * n_steps
   452→        samples_per_sec = 1.0 / time_per_sample_s if time_per_sample_s > 0 else 0
   453→
   454→        # CPU throughput estimate
   455→        cpu_cores = os.cpu_count() or 1
   456→        wall_time_all_samples = optimal_samples / samples_per_sec / cpu_cores if samples_per_sec > 0 else float('inf')
   457→        wall_time_1core = optimal_samples / samples_per_sec if samples_per_sec > 0 else float('inf')
   458→
   459→        # GPU equivalent (assuming 100x speedup over single CPU core)
   460→        gpu_time = wall_time_1core / 100 if wall_time_1core != float('inf') else float('inf')
   461→    else:
   462→        samples_per_sec = 0
   463→        wall_time_1core = float('inf')
   464→        wall_time_all_samples = float('inf')
   465→        gpu_time = float('inf')
   466→
   467→    return {
   468→        "target_params": target_params,
   469→        "best_config": best,
   470→        "architecture": {
   471→            "n_layers": best["n_layers"],
   472→            "n_experts": best["n_experts"],
   473→            "d_model": best["d_model"],
   474→            "d_ff": best["d_ff"],
   475→            "top_k": best["top_k"],
   476→            "active_ratio": best["active_params_ratio"],
   477→            "sparsity": 1 - best["active_params_ratio"],
   478→        },
   479→        "compute": {
   480→            "flops_per_rk4_step": flops_per_step,
   481→            "flops_per_moe_sample": moe_flops_per_sample,
   482→            "flops_per_sample_total": effective_flops,
   483→            "mflops_per_sample": effective_flops / 1e6,
   484→            "mvt_ratio_D_over_N": round(actual_ratio, 1),
   485→            "optimal_samples": int(optimal_samples),
   486→            "chinchilla_transformer_ratio": 20.0,
   487→            "mvt_efficiency_gain": round(expr_mult, 1),
   488→            "chinchilla_transformer_tokens": int(20 * target_params),
   489→            "chinchilla_transformer_flops": 6 * target_params * 20 * target_params,
   490→            "mvt_total_flops": effective_flops * optimal_samples,
   491→        },
   492→        "time_estimates": {
   493→            "samples_per_sec_1core": round(samples_per_sec, 1),
   494→            "wall_time_1core_hours": round(wall_time_1core / 3600, 1),
   495→            "wall_time_all_cores_hours": round(wall_time_all_samples / 3600, 1),
   496→            "gpu_a100_hours": round(gpu_time / 3600, 1),
   497→        },
   498→        "comparison_vs_transformer": {
   499→            "transformer_params": target_params,
   500→            "transformer_optimal_tokens": int(20 * target_params),
   501→            "mvt_params": best["total_params"],
   502→            "mvt_optimal_samples": int(optimal_samples),
   503→            "sample_token_ratio": round(actual_ratio / 20, 2),
   504→            "compute_ratio": round(effective_flops * optimal_samples / (6 * target_params * 20 * target_params), 2),
   505→        },
   506→    }
